#!/usr/bin/env python3
"""Merge raw venue research into data.json for the Thanet food dashboard.

Usage: python3 build_data.py [YYYY-MM-DD]
Inputs (data/raw/*.json + data/venues.json for first-pass metadata).
Output: data/data.json  (keeps first_seen dates + weekly history from any previous data.json)
"""
import json, glob, os, sys, datetime

BASE = os.path.dirname(os.path.abspath(__file__))
RAW = os.path.join(BASE, "raw")
OUT = os.path.join(BASE, "data.json")
today = sys.argv[1] if len(sys.argv) > 1 else datetime.date.today().isoformat()

FOOD_TYPES = {"seasonal", "weekly", "monthly", "specials", "sunday-roast",
              "set-menu", "festive", "event", "main", "other"}

def load(p):
    with open(p, encoding="utf-8") as f:
        return json.load(f)

meta = {v["id"]: v for v in load(os.path.join(BASE, "venues.json"))}
records = []
for path in sorted(glob.glob(os.path.join(RAW, "*.json"))):
    for r in load(path):
        if r["id"] in meta:
            m = meta[r["id"]]
            r = {**m, **r}
        records.append(r)

# Known corrections from research
FIX = {
    "jolly-farmer": {"town": "Manston"},
}

def canon_town(t):
    t = (t or "").strip()
    low = t.lower()
    for name in ("Margate", "Cliftonville", "Ramsgate", "Broadstairs", "Birchington",
                 "Westgate-on-Sea", "Minster", "Monkton", "Manston", "Sarre"):
        if low.startswith(name.lower()):
            return "Margate" if name == "Cliftonville" else name
    if "broadstairs" in low:
        return "Broadstairs"
    return t

# cuisine per venue id; venues not listed (e.g. newly found ones) show as "Not classified"
_cp = os.path.join(BASE, "cuisines.json")
CUISINE = load(_cp) if os.path.exists(_cp) else {}
# FSA food hygiene ratings per venue id (from api.ratings.food.gov.uk, Thanet); missing = not found
_fp = os.path.join(BASE, "fsa.json")
FSA = load(_fp) if os.path.exists(_fp) else {}
_rp = os.path.join(BASE, "reviews.json")  # TripAdvisor ("ta") and OpenTable ("ot") per venue id
REV = load(_rp) if os.path.exists(_rp) else {}

venues, seen = [], set()
for r in records:
    if r["id"] in seen:
        continue
    seen.add(r["id"])
    r.update(FIX.get(r["id"], {}))
    if r.get("trading") == "closed":
        continue
    menus = []
    for m in r.get("menus", []):
        t = m.get("type", "other")
        if t == "drinks":
            continue
        if t not in FOOD_TYPES:
            t = "other"
        menus.append({
            "title": m.get("title", "").strip(),
            "type": t,
            "url": m.get("url"),
            "format": m.get("format", "html"),
            "valid_dates": m.get("valid_dates"),
            "signal": bool(m.get("is_new_signal")),
        })
    notices = r.get("new_menu_notices") or r.get("notices") or []
    venues.append({
        "id": r["id"], "name": r["name"], "town": canon_town(r.get("town", "")),
        "type": r.get("type", "restaurant"), "address": r.get("address", ""),
        "website": r.get("website"), "social": r.get("social"),
        "serves_food": r.get("serves_food", "unclear"),
        "cuisine": CUISINE.get(r["id"], "Not classified"),
        "trading": r.get("trading", "confirmed"),
        "fsa": FSA.get(r["id"]),
        "ta": REV.get(r["id"], {}).get("ta"),
        "ot": REV.get(r["id"], {}).get("ot"),
        "menus": menus, "notices": notices,
        "checked": r.get("checked_on") or today,
    })

# History / first_seen handling
prev = load(OUT) if os.path.exists(OUT) else None
first_seen = {}
history = []
if prev:
    history = prev.get("history", [])
    for v in prev["venues"]:
        for m in v["menus"]:
            first_seen[(v["id"], m["url"], m["title"])] = m.get("first_seen", today)
same_day_baseline = bool(prev and prev.get("baseline") and prev.get("generated") == today)
baseline = (not history) or same_day_baseline
prev_keys = set(first_seen)
# keep flags set earlier the same day, so re-running after adding a venue doesn't erase them
prev_new = {(v["id"], m["url"], m["title"]) for v in (prev or {}).get("venues", [])
            for m in v["menus"] if m.get("is_new_this_week")}
for v in venues:
    for m in v["menus"]:
        k = (v["id"], m["url"], m["title"])
        m["first_seen"] = first_seen.get(k, today)
        if same_day_baseline:
            m["is_new_this_week"] = k not in prev_keys or k in prev_new
        else:
            m["is_new_this_week"] = (not baseline) and m["first_seen"] == today

n_menus = sum(len(v["menus"]) for v in venues)
history = [h for h in history if h["date"] != today]
history.append({
    "date": today,
    "venues": len(venues),
    "menus": n_menus,
    "new_menus": sum(1 for v in venues for m in v["menus"] if m["is_new_this_week"]),
})
history = history[-26:]  # keep ~6 months

out = {
    "generated": today,
    "baseline": baseline and not same_day_baseline or same_day_baseline,
    "run_number": len(history),
    "venues": venues,
    "history": history,
}
with open(OUT, "w", encoding="utf-8") as f:
    json.dump(out, f, ensure_ascii=False, separators=(",", ":"))
print(f"{len(venues)} venues, {n_menus} menus, baseline={baseline}")
